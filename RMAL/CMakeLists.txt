cmake_minimum_required(VERSION 3.20)
project(rmal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rmal_core src/rmal.cpp)
target_include_directories(rmal_core PUBLIC include)
add_executable(rmalc src/main.cpp)
target_link_libraries(rmalc PRIVATE rmal_core)
enable_testing()
add_test(NAME rmal_selfcheck COMMAND rmalc selfcheck)
